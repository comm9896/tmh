		timeout_effect = {

			if = {
				limit = {

			if = {
				limit = {
					not = {
						has_completed_focus = SOV_secure_leningrad
					}

				add_stability = -0.2
				add_war_support = -0.2
				
				declare_war_on = { target = FIN    type = take_state_focus }
			}			
		}
	}
}
