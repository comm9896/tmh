
Sapr_SOV_young_builders_of_communism_level_up = {

	if = {
		limit = { has_idea = SOV_young_builders_of_communism_4 }
		swap_ideas = {
			remove_idea = SOV_young_builders_of_communism_4
			add_idea = SOV_young_builders_of_communism_5
		}
	}
	else_if = {
		limit = { has_idea = SOV_young_builders_of_communism_3 }
		swap_ideas = {
			remove_idea = SOV_young_builders_of_communism_3
			add_idea = SOV_young_builders_of_communism_4
		}
	}
	else_if = {
		limit = { has_idea = SOV_young_builders_of_communism_2 }
		swap_ideas = {
			remove_idea = SOV_young_builders_of_communism_2
			add_idea = SOV_young_builders_of_communism_3
		}
	}
	else_if = {
		limit = { has_idea = SOV_young_builders_of_communism_1 }
		swap_ideas = {
			remove_idea = SOV_young_builders_of_communism_1
			add_idea = SOV_young_builders_of_communism_2
		}		
	}
	else_if = {
		limit = { NOT = { has_idea = SOV_young_builders_of_communism_1 } }
		add_ideas = SOV_young_builders_of_communism_1
	}
}
