add_namespace = Barb_offe_starts

news_event = {
	id = Barb_offe_starts.1
	
	hidden = yes
	fire_only_once = yes
	
	trigger = {			
		and ={
			date > 1941.06.11
			date < 1941.10.01

		}
		
	}	
	

	immediate = {	
		
		for_each_scope_loop = {
				array = global.GER_region_for_barb
				
				add_dynamic_modifier = {
					modifier = Barb_offe
					scope = GER
					days = 135
				}	
				add_dynamic_modifier = {
					modifier = Barb_offe
					scope = ROM
					days = 135
				}	
						
			}		
		
	}		
}
